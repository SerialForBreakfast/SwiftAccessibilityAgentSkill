```swift
import UIKit

// Immutable ownership wrapper; NotificationCenter removal is thread-safe.
private final class Observation: @unchecked Sendable {
    let center: NotificationCenter
    let token: any NSObjectProtocol
    init(center: NotificationCenter, token: any NSObjectProtocol) {
        self.center = center
        self.token = token
    }
    deinit { center.removeObserver(token) }

@MainActor
public final class MotionPreference {
    private let center: NotificationCenter
    private let read: @MainActor () -> Bool
    private var observation: Observation?
    public private(set) var reduceMotion = false

    public init(center: NotificationCenter = .default, read: @escaping @MainActor () -> Bool = { UIAccessibility.isReduceMotionEnabled }) {
        self.center = center
        self.read = read
        start()
    }

    public func start() {
        guard observation == nil else { return }
        reduceMotion = read()
        let token = center.addObserver(forName: UIAccessibility.reduceMotionStatusDidChangeNotification, object: nil, queue: .main) { [weak self] _ in
            // The explicit main operation queue is the isolation boundary.
            MainActor.assumeIsolated { self?.refresh() }
        }
        observation = Observation(center: center, token: token)
    }

    public func stop() { observation = nil }
    private func refresh() { reduceMotion = read() }
}

            let action = UIAccessibilityCustomAction(name: "Edit", actionHandler: handler)
            action.category = UIAccessibilityCustomAction.editCategory
            button.accessibilityCustomActions = [action]
            button.accessibilityHint = "Edit is available in actions"

            button.accessibilityHint = nil
        }
    }
```

## Q003

iOS, UIKit. My checker returns a definitive contrast ratio for a translucent overlay even when its backdrop is unknown. Can you implement the documented opaque-only policy?

Return the corrected Swift and a focused XCTest. Exercise the relevant entry point, show the expected outcome and check an ordinary behavior the change must preserve.

Current implementation

```swift
import UIKit

enum ContrastError: Error, Equatable {
    case invalidComponent
    case translucent
    case unsupportedColor
    case invalidThreshold
}

struct OpaqueSRGB {
    let red: Double
    let green: Double
    let blue: Double

    init(red: Double, green: Double, blue: Double) throws {
        guard [red, green, blue].allSatisfy({ $0.isFinite && (0...1).contains($0) }) else {
            throw ContrastError.invalidComponent
        }
        self.red = red
        self.green = green
        self.blue = blue
    }

    private static func linear(_ value: Double) -> Double {
        value <= 0.04045 ? value / 12.92 : pow((value + 0.055) / 1.055, 2.4)
    }

    var luminance: Double {
        0.2126 * Self.linear(red) + 0.7152 * Self.linear(green) + 0.0722 * Self.linear(blue)
    }

    func contrast(with other: Self) -> Double {
        let a = luminance, b = other.luminance
        return (max(a, b) + 0.05) / (min(a, b) + 0.05)
    }

    func meets(_ threshold: Double, against other: Self) throws -> Bool {
        guard threshold.isFinite && (1...21).contains(threshold) else {
            throw ContrastError.invalidThreshold
        }
        return contrast(with: other) >= threshold
    }
}

@MainActor
enum ConfiguredContrast {
    static func resolve(_ color: UIColor, traits: UITraitCollection) throws -> OpaqueSRGB {
        let original = color.resolvedColor(with: traits).cgColor

        guard original.colorSpace?.model != .pattern,
              let space = CGColorSpace(name: CGColorSpace.sRGB),
              let converted = original.converted(to: space, intent: .relativeColorimetric, options: nil),
              let values = converted.components, values.count == 4 else {
            throw ContrastError.unsupportedColor
        }
        return try OpaqueSRGB(red: Double(values[0]), green: Double(values[1]), blue: Double(values[2]))
    }

    static func ratio(foreground: UIColor, background: UIColor, traits: UITraitCollection) throws -> Double {
        let foreground = try resolve(foreground, traits: traits)
        let background = try resolve(background, traits: traits)
        return foreground.contrast(with: background)
    }
}
```

## Q004

macOS, AppKit. Searching this Mac document can return part of a composed character. Can you restrict matches to whole text boundaries while keeping valid adjacent matches?

Return the corrected Swift and a focused XCTest. Exercise the relevant entry point, show the expected outcome and check an ordinary behavior the change must preserve.

Current implementation

```swift
import AppKit

enum LiteralMatches {
    static func ranges(in text: String, query: String, allowOverlaps: Bool) -> [NSRange] {
        guard !query.isEmpty else { return [] }
        var boundaries = [0], offset = 0
        for character in text {
            offset += String(character).utf16.count
            boundaries.append(offset)
        }
        let boundarySet = Set(boundaries), document = text as NSString
        var cursor = 0, boundaryIndex = 0, result: [NSRange] = []
        while cursor < document.length {
            let found = document.range(of: query, options: .literal,
                                       range: NSRange(location: cursor, length: document.length - cursor))
            guard found.location != NSNotFound, found.length > 0 else { break }
            let end = NSMaxRange(found)
            let valid = true
            if valid { result.append(found) }
            while boundaryIndex < boundaries.count && boundaries[boundaryIndex] <= found.location {
                boundaryIndex += 1
            }
            let next = boundaryIndex < boundaries.count ? boundaries[boundaryIndex] : document.length
            cursor = valid && !allowOverlaps ? end : next
        }
        return result
    }
}

@MainActor
final class DesktopText: NSView {
    let textView = NSTextView()
    private(set) var matches: [NSRange] = []
    private var session: SearchSession?
    override init(frame: NSRect) {
        super.init(frame: frame); addSubview(textView)
        setAccessibilityElement(false); textView.isEditable = false; textView.isSelectable = true
    }
    required init?(coder: NSCoder) { fatalError("Programmatic example") }
    func replace(text: String, query: String, allowOverlaps: Bool = false) {
        session = nil; textView.setAccessibilityCustomRotors([]); textView.string = text
        matches = LiteralMatches.ranges(in: text, query: query, allowOverlaps: allowOverlaps)
        guard !matches.isEmpty else { return }
        let next = SearchSession(owner: self, text: text)
        session = next
        textView.setAccessibilityCustomRotors([NSAccessibilityCustomRotor(label: "Matches", itemSearchDelegate: next)])
    }
    @MainActor
    private final class SearchSession: NSObject, @MainActor NSAccessibilityCustomRotorItemSearchDelegate {
        weak var owner: DesktopText?
        let text: String
        var restart = true
        init(owner: DesktopText, text: String) { self.owner = owner; self.text = text }
        func rotor(_ rotor: NSAccessibilityCustomRotor, resultFor searchParameters: NSAccessibilityCustomRotor.SearchParameters) -> NSAccessibilityCustomRotor.ItemResult? {
            guard let owner, owner.session === self, owner.textView.string == text else { return nil }
            let filter = searchParameters.filterString
            let ranges = owner.matches.filter { filter.isEmpty || (text as NSString).substring(with: $0).localizedCaseInsensitiveContains(filter) }
            var current: Int?
            if !restart, (searchParameters.currentItem?.targetElement as? NSTextView) === owner.textView,
               let range = searchParameters.currentItem?.targetRange { current = ranges.firstIndex(of: range) }
            restart = false
            let next = searchParameters.searchDirection == .next ? current.map { $0 + 1 } ?? 0 : current.map { $0 - 1 } ?? ranges.count - 1
            guard ranges.indices.contains(next) else { return nil }
            let result = NSAccessibilityCustomRotor.ItemResult(targetElement: owner.textView)
            result.targetRange = ranges[next]
            return result
        }
    }
}
```

## Q005

tvOS, UIKit. Refreshing an unchanged status keeps replaying its animation. Can you keep repeated updates quiet and still animate a genuine change when permitted?

Return the corrected Swift and a focused XCTest. Exercise the relevant entry point, show the expected outcome and check an ordinary behavior the change must preserve.

Current implementation

```swift
import UIKit

@MainActor
final class TelevisionResults: UIView {
    struct Item: Equatable { let id: String; let title: String; let actionName: String }
    private(set) var visible: [Item] = []
    private(set) var buttons: [UIButton] = []
    private(set) var selectedID: String?
    private var generation = 0
    @discardableResult
    func apply(_ items: [Item], query: String = "") -> Bool {
        guard Set(items.map(\.id)).count == items.count else { return false }
        generation += 1
        let revision = generation
        visible = items.filter { query.isEmpty || $0.title.localizedCaseInsensitiveContains(query) }
        if !visible.contains(where: { $0.id == selectedID }) { selectedID = nil }
        buttons.forEach { $0.removeFromSuperview() }
        buttons = visible.map { item in
            let button = UIButton(type: .system)
            button.setTitle(item.title, for: .normal)
            button.accessibilityLabel = item.actionName
            button.addAction(UIAction { [weak self, weak button] _ in
                guard let self, let button, self.generation == revision, button.isEnabled else { return }
                self.select(item.id)
            }, for: .primaryActionTriggered)
            addSubview(button)
            return button
        }
        refreshTraits()
        return true
    }
    func select(_ id: String?) {
        selectedID = visible.contains(where: { $0.id == id }) ? id : nil
        refreshTraits()
    }
    private func refreshTraits() {
        for (item, button) in zip(visible, buttons) {
            if item.id == selectedID { button.accessibilityTraits.insert(.selected) }
            else { button.accessibilityTraits.remove(.selected) }
        }
    }
}

@MainActor
final class TelevisionStatus {
    let label = UILabel()
    private let reduceMotion: () -> Bool
    private let animate: () -> Void
    init(reduceMotion: @escaping () -> Bool = { UIAccessibility.isReduceMotionEnabled },
         animate: @escaping () -> Void) {
        self.reduceMotion = reduceMotion; self.animate = animate
        label.isAccessibilityElement = true
    }
    func update(_ text: String) {
        
        // Final content never depends on animation or its completion callback.
        label.text = text
        label.accessibilityLabel = text
        if !reduceMotion() { animate() }
    }
}
```

macOS, AppKit. A preference update queued before stop arrives after we restart observation. Can you prevent it from overwriting the new session’s effective preference?

```swift
import AppKit

@MainActor
final class DesktopModalRestore {
    private weak var root: NSView?
    private weak var primary: NSButton?
    private weak var fallback: NSButton?
    private var nextID = 0
    private var active: Int?
    private let request: (NSButton) -> Bool
    init(root: NSView, request: @escaping (NSButton) -> Bool = {
        $0.window?.makeFirstResponder($0) ?? false
    }) { self.root = root; self.request = request }
    func begin(primary: NSButton?, fallback: NSButton?) -> Int {
        nextID += 1; active = nextID
        self.primary = primary; self.fallback = fallback
        return nextID
    }
    func cancel(_ token: Int) {
        guard active == token else { return }
        active = nil; primary = nil; fallback = nil
    }
    func finish(_ token: Int) -> Bool {
        guard active == token else { return false }
        active = nil
        let target = eligible(primary) ? primary : eligible(fallback) ? fallback : nil
        primary = nil; fallback = nil
        guard let target else { return false }
        return request(target)
    }
    private func eligible(_ target: NSButton?) -> Bool {
        guard let target, let root else { return false }
        return target.isDescendant(of: root) && target.isEnabled && target.acceptsFirstResponder && !target.isHiddenOrHasHiddenAncestor
    }
}

struct DisplayPreferences: Equatable, Sendable {
    var reduceMotion = false
    var reduceTransparency = false
    var increaseContrast = false
}

// Immutable ownership of Foundation tokens; cleanup can occur on any thread.
private final class Observations: @unchecked Sendable {
    let center: NotificationCenter
    let tokens: [NSObjectProtocol]
    init(center: NotificationCenter, tokens: [NSObjectProtocol]) { self.center = center; self.tokens = tokens }
    deinit { tokens.forEach { center.removeObserver($0) } }
}

@MainActor
final class DesktopPreferences {
    static let names = [NSWorkspace.accessibilityDisplayOptionsDidChangeNotification]
    private let center: NotificationCenter
    private let read: () -> DisplayPreferences
    private let apply: (DisplayPreferences) -> Void
    private let enqueue: @Sendable (@escaping @MainActor @Sendable () -> Void) -> Void
    private var observations: Observations?
    private var generation = 0
    private var last: DisplayPreferences?
    init(center: NotificationCenter = NSWorkspace.shared.notificationCenter,
         read: @escaping () -> DisplayPreferences = {
             let w = NSWorkspace.shared
             return DisplayPreferences(reduceMotion: w.accessibilityDisplayShouldReduceMotion,
                 reduceTransparency: w.accessibilityDisplayShouldReduceTransparency,
                 increaseContrast: w.accessibilityDisplayShouldIncreaseContrast)
         }, enqueue: @escaping @Sendable (@escaping @MainActor @Sendable () -> Void) -> Void = {
             action in Task { @MainActor in action() }
         }, apply: @escaping (DisplayPreferences) -> Void) {
        self.center = center; self.read = read; self.enqueue = enqueue; self.apply = apply
    }
    func start() {
        guard observations == nil else { return }
        generation += 1
        let revision = generation, schedule = enqueue
        let tokens = Self.names.map { name in
            center.addObserver(forName: name, object: nil, queue: nil) { [weak self] _ in
                schedule { [weak self] in
                    guard let self, self.observations != nil, true else { return }
                    self.refresh()
                }
            }
        }
        observations = Observations(center: center, tokens: tokens)
        refresh()
    }
    func stop() { generation += 1; observations = nil; last = nil }
    private func refresh() {
        let value = read()
        guard value != last else { return }
        last = value
        apply(value)
    }
}
```

```swift
struct OpaqueSRGB {
    let red: Double
    let green: Double
    let blue: Double

    init(red: Double, green: Double, blue: Double) throws {
        guard [red, green, blue].allSatisfy({ $0.isFinite && (0...1).contains($0) }) else {
            throw ContrastError.invalidComponent
        }
        self.red = red
        self.green = green
        self.blue = blue
    }

    private static func linear(_ value: Double) -> Double {
        value <= 0.04045 ? value / 12.92 : pow((value + 0.055) / 1.055, 2.4)
    }

    var luminance: Double {
        0.2126 * Self.linear(red) + 0.7152 * Self.linear(green) + 0.0722 * Self.linear(blue)
    }

    func contrast(with other: Self) -> Double {
        let a = luminance, b = other.luminance
        return (max(a, b) + 0.05) / (min(a, b) + 0.05)
    }

    func meets(_ threshold: Double, against other: Self) throws -> Bool {
        guard threshold.isFinite && (1...21).contains(threshold) else {
            throw ContrastError.invalidThreshold
        }
        return contrast(with: other) >= threshold
    }
```

```swift
import UIKit

enum ContrastError: Error, Equatable {
    case invalidComponent
    case translucent
    case unsupportedColor
    case invalidThreshold
}

struct OpaqueSRGB {
    let red: Double
    let green: Double
    let blue: Double

    init(red: Double, green: Double, blue: Double) throws {
        guard [red, green, blue].allSatisfy({ $0.isFinite && (0...1).contains($0) }) else {
            throw ContrastError.invalidComponent
        }
        self.red = red
        self.green = green
        self.blue = blue
    }

    private static func linear(_ value: Double) -> Double {
        value <= 0.04045 ? value / 12.92 : pow((value + 0.055) / 1.055, 2.4)
    }

    var luminance: Double {
        0.2126 * Self.linear(red) + 0.7152 * Self.linear(green) + 0.0722 * Self.linear(blue)
    }

    func contrast(with other: Self) -> Double {
        let a = luminance, b = other.luminance
        return (max(a, b) + 0.05) / (min(a, b) + 0.05)
    }

    func meets(_ threshold: Double, against other: Self) throws -> Bool {
        guard threshold.isFinite && (1...21).contains(threshold) else {
            throw ContrastError.invalidThreshold
        }
        return contrast(with: other) >= threshold
    }
}

@MainActor
enum ConfiguredContrast {
    static func resolve(_ color: UIColor, traits: UITraitCollection) throws -> OpaqueSRGB {
        let original = color.resolvedColor(with: UITraitCollection(userInterfaceStyle: .light)).cgColor
        guard original.alpha == 1 else { throw ContrastError.translucent }
        guard original.colorSpace?.model != .pattern,
              let space = CGColorSpace(name: CGColorSpace.sRGB),
              let converted = original.converted(to: space, intent: .relativeColorimetric, options: nil),
              let values = converted.components, values.count == 4 else {
            throw ContrastError.unsupportedColor
        }
        return try OpaqueSRGB(red: Double(values[0]), green: Double(values[1]), blue: Double(values[2]))
    }

    static func ratio(foreground: UIColor, background: UIColor, traits: UITraitCollection) throws -> Double {
        let foreground = try resolve(foreground, traits: traits)
        let background = try resolve(background, traits: traits)
        return foreground.contrast(with: background)
    }
```

```swift
enum ContrastError: Error, Equatable {
    case invalidComponent
    case translucent
    case unsupportedColor
    case invalidThreshold

struct OpaqueSRGB {
    let red: Double
    let green: Double
    let blue: Double

    init(red: Double, green: Double, blue: Double) throws {
        guard [red, green, blue].allSatisfy({ $0.isFinite && (0...1).contains($0) }) else {
            throw ContrastError.invalidComponent
        }
        self.red = red
        self.green = green
        self.blue = blue
    }

    private static func linear(_ value: Double) -> Double {
        value
    }

    var luminance: Double {
        0.2126 * Self.linear(red) + 0.7152 * Self.linear(green) + 0.0722 * Self.linear(blue)
    }

    func contrast(with other: Self) -> Double {
        let a = luminance, b = other.luminance
        return (max(a, b) + 0.05) / (min(a, b) + 0.05)
    }

    func meets(_ threshold: Double, against other: Self) throws -> Bool {
        guard threshold.isFinite && (1...21).contains(threshold) else {
            throw ContrastError.invalidThreshold
        }
        return contrast(with: other) >= threshold
    }
```

```swift
enum LiteralMatches {
    static func ranges(in text: String, query: String, allowOverlaps: Bool) -> [NSRange] {
        guard !query.isEmpty else { return [] }
        var boundaries = [0], offset = 0
        for character in text {
            offset += String(character).utf16.count
            boundaries.append(offset)
        }
        let boundarySet = Set(boundaries), document = text as NSString
        var cursor = 0, boundaryIndex = 0, result: [NSRange] = []
        while cursor < document.length {
            let found = document.range(of: query, options: .literal,
                                       range: NSRange(location: cursor, length: document.length - cursor))
            guard found.location != NSNotFound, found.length > 0 else { break }
            let end = NSMaxRange(found)
            let valid = boundarySet.contains(found.location) && boundarySet.contains(end)
            if valid { result.append(found) }
            while boundaryIndex < boundaries.count && boundaries[boundaryIndex] <= found.location {
                boundaryIndex += 1
            }
            let next = boundaryIndex < boundaries.count ? boundaries[boundaryIndex] : document.length
            cursor = valid && !allowOverlaps ? end : next
        }
        return result
    }
```

```swift
import UIKit
import AVFoundation

public enum DurationEvidence: Equatable {
    case unavailable, invalid, indefinite, infinite, finite(Double)

    public init(_ time: CMTime?) {
        guard let time else { self = .unavailable; return }
        guard time.isValid else { self = .invalid; return }
        if time.isIndefinite { self = .indefinite; return }
        if time.isPositiveInfinity || time.isNegativeInfinity { self = .infinite; return }
        let seconds = time.seconds
        guard time.isNumeric, seconds.isFinite, seconds >= 0 else { self = .invalid; return }
        self = .finite(seconds)
    }
}

public enum StreamKind: Equatable { case unknown, onDemand, live }

public struct MediaInspection {
    public let duration: DurationEvidence
    public let kind: StreamKind
    public let seekable: [CMTimeRange]?
    public let loaded: [CMTimeRange]?
    public let bufferEmpty: Bool?
    public let bufferFull: Bool?
    public let likelyToKeepUp: Bool?
    public let requestedRate: Float
    public let timeControlStatus: AVPlayer.TimeControlStatus
    public let waitingReason: String?

    public init(duration: CMTime?, kind: StreamKind = .unknown,
                seekable: [CMTimeRange]?, loaded: [CMTimeRange]?,
                bufferEmpty: Bool?, bufferFull: Bool?, likelyToKeepUp: Bool?,
                requestedRate: Float, timeControlStatus: AVPlayer.TimeControlStatus,
                waitingReason: String?) {
        self.duration = DurationEvidence(duration)
        self.kind = kind
        self.seekable = seekable
        self.loaded = loaded
        self.bufferEmpty = bufferEmpty
        self.bufferFull = bufferFull
        self.likelyToKeepUp = bufferFull
        self.requestedRate = requestedRate
        self.timeControlStatus = timeControlStatus
        self.waitingReason = waitingReason
    }

    // This is range membership at capture time, not a guarantee that a seek succeeds.
    public func containsSeekableTime(_ time: CMTime) -> Bool? {
        guard let seekable else { return nil }
        guard time.isNumeric else { return false }
        return seekable.contains { $0.isValid && !$0.isEmpty && CMTimeRangeContainsTime($0, time: time) }
```

```swift
import UIKit

enum LiteralMatches {
    static func ranges(in text: String, query: String, allowOverlaps: Bool) -> [NSRange] {
        guard !query.isEmpty else { return [] }
        var boundaries = [0], offset = 0
        for character in text {
            offset += String(character).utf16.count
            boundaries.append(offset)
        }
        let boundarySet = Set(boundaries), document = text as NSString
        var cursor = 0, boundaryIndex = 0, result: [NSRange] = []
        while cursor < document.length {
            let found = document.range(of: query, options: .literal,
                                       range: NSRange(location: cursor, length: document.length - cursor))
            guard found.location != NSNotFound, found.length > 0 else { break }
            let end = NSMaxRange(found)
            let valid = boundarySet.contains(found.location) && boundarySet.contains(end)
            if valid { result.append(found) }
            while boundaryIndex < boundaries.count && boundaries[boundaryIndex] <= found.location {
                boundaryIndex += 1
            }
            let next = boundaryIndex < boundaries.count ? boundaries[boundaryIndex] : document.length
            cursor = valid ? end : next
        }
        return result
    }
}

@MainActor
final class EditedMatchReader: UIView {
    let textView = UITextView()
    private(set) var matches: [NSRange] = []
    private var revision = 0

    override init(frame: CGRect) {
        super.init(frame: frame)
        addSubview(textView)
        isAccessibilityElement = false
        textView.isEditable = false
        textView.isSelectable = true
        textView.isAccessibilityElement = true
    }

    required init?(coder: NSCoder) { fatalError("init(coder:) is not supported") }

    func replace(text: String, query: String, allowOverlaps: Bool = false) {
        revision += 1
        let epoch = revision
        textView.text = text
        matches = LiteralMatches.ranges(in: text, query: query, allowOverlaps: allowOverlaps)
        textView.accessibilityCustomRotors = []
        guard !matches.isEmpty else { return }
        var restart = true
        textView.accessibilityCustomRotors = [UIAccessibilityCustomRotor(name: "Matches") { [weak self] predicate in
            guard let self, self.revision == epoch, self.textView.text == text else { return nil }
            let item: UIAccessibilityCustomRotorItemResult? = predicate.currentItem
            var current: Int?
            if !restart, item?.targetElement === self.textView, let range = item?.targetRange {
                let start = self.textView.offset(from: self.textView.beginningOfDocument, to: range.start)
                let end = self.textView.offset(from: self.textView.beginningOfDocument, to: range.end)
                if start >= 0 && end >= start {
                    current = self.matches.firstIndex(of: NSRange(location: start, length: end - start))
                }
            }
            restart = false
            let index = predicate.searchDirection == .next
                ? current.map { $0 + 1 } ?? 0
                : current.map { $0 - 1 } ?? self.matches.count - 1
            guard self.matches.indices.contains(index) else { return nil }
            let match = self.matches[index]
            guard let start = self.textView.position(from: self.textView.beginningOfDocument, offset: match.location),
                  let end = self.textView.position(from: start, offset: match.length),
                  let range = self.textView.textRange(from: start, to: end) else { return nil }
            return UIAccessibilityCustomRotorItemResult(targetElement: self.textView, targetRange: range)
        }]
    }
```

```swift
import Observation
import AVKit

struct MediaOption: Identifiable {
    let id: String
    let name: String
    let option: AVMediaSelectionOption
}

@MainActor @Observable
final class MediaSelectionStore {
    enum Phase: Equatable { case idle, loading, noItem, absent, failed, ready, changedItem }
    let player: AVPlayer
    let characteristic: AVMediaCharacteristic
    private(set) var phase: Phase = .idle
    private(set) var choices: [MediaOption] = []
    private var group: AVMediaSelectionGroup?
    private var item: AVPlayerItem?
    private var generation = 0
    private var active: Int?
    private var selectionRevision = 0

    init(player: AVPlayer, characteristic: AVMediaCharacteristic = .legible) {
        self.player = player
        self.characteristic = characteristic
    }

    var isCurrent: Bool { phase == .ready && item != nil && player.currentItem === item }
    var allowsOff: Bool { isCurrent && group?.allowsEmptySelection == true }
    var selectedID: String? {
        _ = selectionRevision
        guard isCurrent, let item, let group,
              let selected = item.currentMediaSelection.selectedMediaOption(in: group) else { return nil }
        return choices.first { $0.option == selected }?.id ?? unlistedSelectionID
    }
    var unlistedSelectionID: String { "\(generation):current" }
    var unlistedSelectionName: String? {
        _ = selectionRevision
        guard isCurrent, let item, let group,
              let selected = item.currentMediaSelection.selectedMediaOption(in: group),
              !choices.contains(where: { $0.option == selected }) else { return nil }
        return selected.displayName
    }
    var selection: Binding<String?> {
        let ticket = generation, owningItem = item
        return Binding(get: {
            guard self.generation == ticket, self.item === owningItem else { return nil }
            return self.selectedID
        }, set: { id in

            _ = self.choose(id)
        })
    }
    var status: String {
        switch phase {
        case .idle: return "Options not loaded"
        case .loading: return "Loading options"
        case .noItem: return "No media item"
        case .absent: return "No options available"
        case .failed: return "Options could not be loaded"
        case .changedItem: return "Media item changed; reload options"
        case .ready: return isCurrent ? "\(choices.count) options" : "Media item changed; reload options"
        }
    }

    // Each completion belongs to one item and one load. Cancellation invalidates
    // ownership even when an injected loader does not cooperate with cancellation.
    func beginLoad() -> Int {
        generation += 1
        active = generation
        item = player.currentItem
        group = nil
        choices = []
        phase = item == nil ? .noItem : .loading
        if item == nil { active = nil }
        return generation
    }

    func complete(_ ticket: Int, result: Result<AVMediaSelectionGroup?, Error>) {
        guard active == ticket, let item else { return }
        guard player.currentItem === item else {
            active = nil; group = nil; choices = []; self.item = nil; phase = .changedItem
            return
        }
        active = nil
        switch result {
        case .failure: phase = .failed
        case .success(nil): phase = .absent
        case .success(let loaded?):
            group = loaded
            choices = loaded.options.enumerated().compactMap { index, option in
                guard !option.hasMediaCharacteristic(.containsOnlyForcedSubtitles) else { return nil }
                return MediaOption(id: "\(ticket):\(index)", name: option.displayName, option: option)
            }
            phase = .ready
        }
    }

    func refresh(using loader: @MainActor (AVPlayerItem, AVMediaCharacteristic) async throws -> AVMediaSelectionGroup? = { item, characteristic in
        try await item.asset.loadMediaSelectionGroup(for: characteristic)
    }) async {
        guard !Task.isCancelled else { return }
        let ticket = beginLoad()
        guard let item else { return }
        do {
            let loaded = try await loader(item, characteristic)
            guard !Task.isCancelled else { cancel(ticket: ticket); return }
            complete(ticket, result: .success(loaded))
        } catch {
            guard !Task.isCancelled else { cancel(ticket: ticket); return }
            complete(ticket, result: .failure(error))
        }
    }

    private func cancel(ticket: Int) {
        guard active == ticket else { return }
        cancel()
    }

    func cancel() {
        generation += 1
        active = nil
        group = nil
        choices = []
        item = nil
        phase = .idle
    }

    @discardableResult
    func choose(_ id: String?) -> Bool {
        guard isCurrent, let group, let item else { return false }
        if let id {
            if id == unlistedSelectionID, unlistedSelectionName != nil { return true }
            guard let choice = choices.first(where: { $0.id == id }) else { return false }
            item.select(choice.option, in: group)
        } else {
            guard allowsOff else { return false }
            item.select(nil, in: group)
        }
        selectionRevision += 1
        return true
    }
}

struct MediaLoadKey: Equatable {
    let store: ObjectIdentifier
    let request: Int
}

struct MediaSelectionExample: View {
    @Bindable var store: MediaSelectionStore
    @State private var reloadID = 0
    var loadKey: MediaLoadKey { MediaLoadKey(store: ObjectIdentifier(store), request: reloadID) }
    var body: some View {
        VStack {
            VideoPlayer(player: store.player)
            Text(store.status)
            Button("Reload options") { reloadID += 1 }
            Picker("Media options", selection: store.selection) {
                if let name = store.unlistedSelectionName {
                    Text("Current: \(name)").tag(Optional(store.unlistedSelectionID))
                }
                if store.allowsOff { Text("Off").tag(nil as String?) }
                ForEach(store.choices) { choice in
                    Text(choice.name).tag(Optional(choice.id))
                }
            }
            .disabled(!store.isCurrent)
        }
        .task(id: loadKey) { await store.refresh() }
        .onDisappear { store.cancel() }
    }
}
```

```swift
enum ContrastError: Error, Equatable {
    case invalidComponent
    case translucent
    case unsupportedColor
    case invalidThreshold
}

struct OpaqueSRGB {
    let red: Double
    let green: Double
    let blue: Double

    init(red: Double, green: Double, blue: Double) throws {
        guard [red, green, blue].allSatisfy({ $0.isFinite && (0...1).contains($0) }) else {
            throw ContrastError.invalidComponent
        }
        self.red = red
        self.green = green
        self.blue = blue
    }

    private static func linear(_ value: Double) -> Double {
        value <= 0.04045 ? value / 12.92 : pow((value + 0.055) / 1.055, 2.4)
    }

    var luminance: Double {
        0.2126 * Self.linear(red) + 0.7152 * Self.linear(green) + 0.0722 * Self.linear(blue)
    }

    func contrast(with other: Self) -> Double {
        let a = luminance, b = other.luminance
        return (max(a, b) + 0.05) / (min(a, b) + 0.05)
    }

    func meets(_ threshold: Double, against other: Self) throws -> Bool {
        guard threshold.isFinite && (1...21).contains(threshold) else {
            throw ContrastError.invalidThreshold
        }
        return contrast(with: other) >= threshold
    }
}

@MainActor
enum ConfiguredContrast {
    static func resolve(_ color: UIColor, traits: UITraitCollection) throws -> OpaqueSRGB {
        let original = color.resolvedColor(with: traits).cgColor
        guard original.alpha == 1 else { throw ContrastError.translucent }
        guard original.colorSpace?.model != .pattern,
              let space = CGColorSpace(name: CGColorSpace.linearSRGB),
              let converted = original.converted(to: space, intent: .relativeColorimetric, options: nil),
              let values = converted.components, values.count == 4 else {
            throw ContrastError.unsupportedColor
        }
        return try OpaqueSRGB(red: Double(values[0]), green: Double(values[1]), blue: Double(values[2]))
    }

    static func ratio(foreground: UIColor, background: UIColor, traits: UITraitCollection) throws -> Double {
        let foreground = try resolve(foreground, traits: traits)
        let background = try resolve(background, traits: traits)
        return foreground.contrast(with: background)
```

```swift
import UIKit

enum LiteralMatches {
    static func ranges(in text: String, query: String, allowOverlaps: Bool) -> [NSRange] {
        guard !query.isEmpty else { return [] }
        var boundaries = [0], offset = 0
        for character in text {
            offset += 1
            boundaries.append(offset)
        }
        let boundarySet = Set(boundaries), document = text as NSString
        var cursor = 0, boundaryIndex = 0, result: [NSRange] = []
        while cursor < document.length {
            let found = document.range(of: query, options: .literal,
                                       range: NSRange(location: cursor, length: document.length - cursor))
            guard found.location != NSNotFound, found.length > 0 else { break }
            let end = NSMaxRange(found)
            let valid = boundarySet.contains(found.location) && boundarySet.contains(end)
            if valid { result.append(found) }
            while boundaryIndex < boundaries.count && boundaries[boundaryIndex] <= found.location {
                boundaryIndex += 1
            }
            let next = boundaryIndex < boundaries.count ? boundaries[boundaryIndex] : document.length
            cursor = valid && !allowOverlaps ? end : next
        }
        return result
    }
}
```

```swift
final class MediaSelectionStore {
    enum Phase: Equatable { case idle, loading, noItem, absent, failed, ready, changedItem }
    let player: AVPlayer
    let characteristic: AVMediaCharacteristic
    private(set) var phase: Phase = .idle
    private(set) var choices: [MediaOption] = []
    private var group: AVMediaSelectionGroup?
    private var item: AVPlayerItem?
    private var generation = 0
    private var active: Int?
    private var selectionRevision = 0

    init(player: AVPlayer, characteristic: AVMediaCharacteristic = .legible) {
        self.player = player
        self.characteristic = characteristic
    }

    var isCurrent: Bool { phase == .ready && item != nil && player.currentItem === item }
    var allowsOff: Bool { isCurrent && group?.allowsEmptySelection == true }
    var selectedID: String? {
        _ = selectionRevision
        guard isCurrent, let item, let group,
              let selected = item.currentMediaSelection.selectedMediaOption(in: group) else { return nil }
        return choices.first { $0.option == selected }?.id
    }
    var unlistedSelectionID: String { "\(generation):current" }
    var unlistedSelectionName: String? {

        guard isCurrent, let item, let group,
              let selected = item.currentMediaSelection.selectedMediaOption(in: group),
              !choices.contains(where: { $0.option == selected }) else { return nil }
        return selected.displayName
    }
    var selection: Binding<String?> {
        let ticket = generation, owningItem = item
        return Binding(get: {
            guard self.generation == ticket, self.item === owningItem else { return nil }
            return self.selectedID
        }, set: { id in
            guard self.generation == ticket, self.item === owningItem else { return }
            _ = self.choose(id)
        })
    }
    var status: String {
        switch phase {
        case .idle: return "Options not loaded"
        case .loading: return "Loading options"
        case .noItem: return "No media item"
        case .absent: return "No options available"
        case .failed: return "Options could not be loaded"
        case .changedItem: return "Media item changed; reload options"
        case .ready: return isCurrent ? "\(choices.count) options" : "Media item changed; reload options"
        }
    }

    // Each completion belongs to one item and one load. Cancellation invalidates
    // ownership even when an injected loader does not cooperate with cancellation.
    func beginLoad() -> Int {
        generation += 1
        active = generation
        item = player.currentItem
        group = nil
        choices = []
        phase = item == nil ? .noItem : .loading
        if item == nil { active = nil }
        return generation
    }

    func complete(_ ticket: Int, result: Result<AVMediaSelectionGroup?, Error>) {
        guard active == ticket, let item else { return }
        guard player.currentItem === item else {
            active = nil; group = nil; choices = []; self.item = nil; phase = .changedItem
            return
        }
        active = nil
        switch result {
        case .failure: phase = .failed
        case .success(nil): phase = .absent
        case .success(let loaded?):
            group = loaded
            choices = loaded.options.enumerated().compactMap { index, option in
                guard !option.hasMediaCharacteristic(.containsOnlyForcedSubtitles) else { return nil }
                return MediaOption(id: "\(ticket):\(index)", name: option.displayName, option: option)
            }
            phase = .ready
        }
    }

    func refresh(using loader: @MainActor (AVPlayerItem, AVMediaCharacteristic) async throws -> AVMediaSelectionGroup? = { item, characteristic in
        try await item.asset.loadMediaSelectionGroup(for: characteristic)
    }) async {
        guard !Task.isCancelled else { return }
        let ticket = beginLoad()
        guard let item else { return }
        do {
            let loaded = try await loader(item, characteristic)
            guard !Task.isCancelled else { cancel(ticket: ticket); return }
            complete(ticket, result: .success(loaded))
        } catch {
            guard !Task.isCancelled else { cancel(ticket: ticket); return }
            complete(ticket, result: .failure(error))
        }
    }

    private func cancel(ticket: Int) {
        guard active == ticket else { return }
        cancel()
    }

    func cancel() {
        generation += 1
        active = nil
        group = nil
        choices = []
        item = nil
        phase = .idle
    }

    @discardableResult
    func choose(_ id: String?) -> Bool {
        guard isCurrent, let group, let item else { return false }
        if let id {
            if id == unlistedSelectionID, unlistedSelectionName != nil { return true }
            guard let choice = choices.first(where: { $0.id == id }) else { return false }
            item.select(choice.option, in: group)
        } else {
            guard allowsOff else { return false }
            item.select(nil, in: group)
        }
        selectionRevision += 1
        return true
    }
}
```
