const cases=[];

 const families=[
 ['media-selection','MediaChoiceTests',`Our ${platform} app uses AVPlayerItem. Add helpers for explicit language selection, optionally requiring audio description, and for turning a group off when allowed.  'Use current group options and locale language metadata; exclude forced-only options for full-subtitle requests, require the describes-video characteristic when requested. Load asset metadata asynchronously. Distinguish no current item, a successfully loaded but absent group, and a load error. Inspecting must not change the chosen audio or subtitle-off state. Show a testable implementation targeting ${platform}26 and explain what a tagged silent local test asset can establish.`,
 'Query the actual media characteristics rather than infer description from alternate language or option name. Preserve existing audio and subtitle-off selection. Load duration/group asynchronously, retain unavailable versus absent and propagate errors. Metadata loading is not readyToPlay or narration quality. Accept typed result errors or throwing APIs; no requirement for the example enum names.'],
 ['media-inspection','MediaInspectionTests',`We are adding a diagnostic snapshot to our ${platform} AVPlayer app. Capture duration, loaded and seekable ranges, buffer flags, rate, time-control status and waiting reason without modifying playback.  'Keep unavailable, invalid, indefinite, infinite and finite duration evidence distinct. Unknown stream classification remains unknown: indefinite is not proof of live.  ];
 for(const [topic,suite,prompt,criteria] of families){
 const p=root+'/evaluation/'+topic+'.prompt.md',r=root+'/evaluation/'+topic+'.rubric.md';
 fs.writeFileSync(p,prompt+'\n');

 cases.push({id:folder+'/'+topic,lifecycle:'draft-not-run',subject:{path:p,sha256:hash(fs.readFileSync(p)),text:fs.readFileSync(p,'utf8')},outputs:{status:'not-run',attempts:[]},analysis:{path:r,sha256:hash(fs.readFileSync(r)),answerAnalysis:null},nativeManifest:root+'/manifest.json',authoringCapture:'AUTHORING.md; exact visible request, authoring source and native artifacts; full provider context unavailable'});
 }
}
fs.writeFileSync('research/evaluation/platform-media-v1/prompt-records.json',JSON.stringify({schema:1,scope:'Six development cases; separate from the nine-case Gemini plan and four lifecycle cases',cases},null,2)+'\n');
